cmake_minimum_required(VERSION 3.25)
project(modeld_trt_comparison LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
set(SDK_ROOT "" CACHE PATH "TensorRT or TensorRT-RTX SDK directory")
set(BACKEND "rtx" CACHE STRING "rtx or trt")
if(BACKEND STREQUAL "rtx")
  set(INFERENCE_LIBRARY tensorrt_rtx)
  set(PARSER_LIBRARY tensorrt_onnxparser_rtx)
elseif(BACKEND STREQUAL "trt")
  set(INFERENCE_LIBRARY nvinfer)
  set(PARSER_LIBRARY nvonnxparser)
else()
  message(FATAL_ERROR "BACKEND must be rtx or trt")
endif()
find_library(TRT_LIBRARY ${INFERENCE_LIBRARY} PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ONNX_LIBRARY ${PARSER_LIBRARY} PATHS "${SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(trt-bench bench.cpp ../../src/kernels.cu)
target_include_directories(trt-bench SYSTEM PRIVATE "${SDK_ROOT}/include")
target_compile_definitions(trt-bench PRIVATE BENCH_BACKEND="${BACKEND}")
target_link_libraries(trt-bench PRIVATE CUDA::cudart ${TRT_LIBRARY} ${ONNX_LIBRARY} OpenSSL::Crypto nlohmann_json::nlohmann_json)
set_target_properties(trt-bench PROPERTIES BUILD_RPATH "${SDK_ROOT}/lib")
add_executable(compare-output compare.cpp)
target_link_libraries(compare-output PRIVATE nlohmann_json::nlohmann_json OpenMP::OpenMP_CXX)
