cmake_minimum_required(VERSION 3.25)
project(modeld VERSION 2.1.6 LANGUAGES CXX)
include(GNUInstallDirs)
option(MODELD_CLIENT "Build modelctl and the native mpv plugin" ON)
option(MODELD_BROWSER "Build the browser native messaging host" OFF)
option(MODELD_SERVER "Build the CUDA/TensorRT-RTX renderer" OFF)
option(BUILD_TESTING "Build tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
add_library(modeld_core STATIC src/core.cpp src/protocol.cpp src/log.cpp src/fmp4.cpp)
target_compile_definitions(modeld_core PUBLIC MODELD_VERSION="${PROJECT_VERSION}")
target_include_directories(modeld_core PUBLIC include)
target_link_libraries(modeld_core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
target_compile_options(modeld_core PRIVATE -Wall -Wextra -Wpedantic)
if(MODELD_CLIENT OR MODELD_BROWSER OR BUILD_TESTING)
  find_package(httplib CONFIG REQUIRED)
  add_library(modeld_client STATIC src/config.cpp src/api.cpp src/source.cpp src/media_proxy.cpp src/mpv_source.cpp src/bilibili.cpp src/session.cpp)
  target_link_libraries(modeld_client PUBLIC modeld_core httplib::httplib)
  target_compile_options(modeld_client PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MODELD_CLIENT)
  pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
  add_executable(modelctl src/modelctl.cpp)
  target_link_libraries(modelctl PRIVATE modeld_client)
  add_library(modeld_mpv MODULE src/mpv_plugin.cpp src/mpv_state.cpp)
  set_target_properties(modeld_mpv PROPERTIES PREFIX "" OUTPUT_NAME "modeld")
  target_link_libraries(modeld_mpv PRIVATE modeld_client PkgConfig::MPV)
  target_compile_options(modeld_mpv PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS modelctl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(TARGETS modeld_mpv LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/mpv/scripts)
  install(FILES config/client.example.json DESTINATION ${CMAKE_INSTALL_DATADIR}/modeld)
endif()
if(MODELD_BROWSER)
  add_executable(modeld-browser-host src/browser_host.cpp)
  target_link_libraries(modeld-browser-host PRIVATE modeld_client)
  target_compile_options(modeld-browser-host PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS modeld-browser-host RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(DIRECTORY browser/ DESTINATION ${CMAKE_INSTALL_DATADIR}/modeld/browser)
endif()
if(MODELD_SERVER)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS /opt/tensorrt-rtx/include ENV TENSORRT_ROOT PATH_SUFFIXES include)
  find_library(TENSORRT_LIBRARY tensorrt_rtx HINTS /opt/tensorrt-rtx/lib ENV TENSORRT_ROOT PATH_SUFFIXES lib lib64)
  find_library(TENSORRT_PARSER tensorrt_onnxparser_rtx HINTS /opt/tensorrt-rtx/lib ENV TENSORRT_ROOT PATH_SUFFIXES lib lib64)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT TENSORRT_PARSER)
    message(FATAL_ERROR "TensorRT-RTX SDK is required")
  endif()
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
  add_library(modeld_rife STATIC src/rife.cpp src/kernels.cu)
  set_target_properties(modeld_rife PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_include_directories(modeld_rife PUBLIC include PRIVATE ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(modeld_rife PUBLIC CUDA::cudart CUDA::cuda_driver ${TENSORRT_LIBRARY} ${TENSORRT_PARSER} OpenSSL::Crypto)
  add_library(modeld_native STATIC src/native_api.cpp src/media.cpp)
  target_link_libraries(modeld_native PRIVATE modeld_core modeld_rife PkgConfig::AV)
  target_compile_options(modeld_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  enable_testing()
  add_executable(modeld-tests tests/protocol.cpp)
  target_link_libraries(modeld-tests PRIVATE modeld_client)
  add_test(NAME protocol COMMAND modeld-tests)
  set_tests_properties(protocol PROPERTIES ENVIRONMENT "XDG_STATE_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-state")
  if(MODELD_SERVER)
    add_executable(modeld-fault-source tests/fault_source.cpp)
    target_link_libraries(modeld-fault-source PRIVATE modeld_client)
  endif()
endif()
